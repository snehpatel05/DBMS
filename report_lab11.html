const { Pool } = require('pg');
require('dotenv').config();

const pool = new Pool({
    connectionString: process.env.DATABASE_URL,
    ssl: { rejectUnauthorized: false }
});

2. Centralized Query Helper Function (server.js)

// Centralized Query Helper with execution duration and row count logging
async function query(text, params) {
    const start = Date.now();
    const res = await pool.query(text, params);
    const duration = Date.now() - start;
    console.log(`[SQL EXEC] executed query: ${text.replace(/\s+/g, ' ').trim()} | duration: ${duration}ms | rows: ${res.rowCount}`);
    return res;
}

B.2 READ (SELECT) Operation

SQL 
SELECT c.cnr, c.standardized_filing_number, c.case_number, c.fir_number,
       c.statutory_urgency, c.case_status, c.registered_by_clerk_id,
       u.name AS registered_by_clerk_name
FROM court_case c
LEFT JOIN system_users u ON c.registered_by_clerk_id = u.user_id
WHERE c.cnr = $1;
Function 
app.get('/api/cases', async (req, res) => {
    const { search, status, urgency } = req.query;
    let conditions = [], params = [];
    if (search) {
        params.push(`%${search.trim()}%`);
        conditions.push(`(c.cnr ILIKE $${params.length} OR c.case_number ILIKE $${params.length})`);
    }
    const whereSql = conditions.length > 0 ? `WHERE ${conditions.join(' AND ')}` : '';
    const sql = `SELECT c.*, u.name as registered_by_clerk_name FROM court_case c 
                 LEFT JOIN system_users u ON c.registered_by_clerk_id = u.user_id 
                 ${whereSql} ORDER BY c.cnr DESC LIMIT 100;`;
    const result = await query(sql, params);
    res.json({ success: true, rowCount: result.rowCount, data: result.rows });
});

B.3 UPDATE Operation

SQL 
UPDATE court_case 
SET statutory_urgency = $1, case_status = $2 
WHERE cnr = $3 
RETURNING *;
Function 
app.put('/api/cases/:cnr', async (req, res) => {
    const { cnr } = req.params;
    const { urgency, status } = req.body;
    const sql = `UPDATE court_case 
                 SET statutory_urgency = COALESCE($1, statutory_urgency),
                     case_status = COALESCE($2, case_status)
                 WHERE cnr = $3 RETURNING *;`;
    const params = [urgency || null, status || null, cnr];
    const result = await query(sql, params);
    if (result.rowCount === 0) return res.status(404).json({ error: 'Case not found' });
    res.json({ success: true, rowCount: result.rowCount, data: result.rows[0] });
});

SQL

Function 
app.delete('/api/cases/:cnr', async (req, res) => {
    const { cnr } = req.params;
    const userRole = req.headers['x-user-role'];
    if (userRole && !['Clerk', 'Court Administrator'].includes(userRole)) {
        return res.status(403).json({ error: 'Only registry clerks can delete records.' });
    }
    const sql = `DELETE FROM court_case WHERE cnr = $1 RETURNING *;`;
    const result = await query(sql, [cnr]);
    if (result.rowCount === 0) return res.status(404).json({ error: 'Case not found' });
    res.json({ success: true, rowCount: result.rowCount, message: 'Case deleted' });
});

(D) Reflection & Conclusion

During the development of Lab 11 for our District Court Case Management System, our team navigated several critical real-world database connectivity challenges. The primary hurdle was network routing to Supabase PostgreSQL: the default hostname only resolved to an IPv6 address, which failed on our local network with ENOTFOUND socket errors. We resolved this by identifying and configuring the dedicated IPv4 transaction pooler located in the AWS Tokyo region (aws-0-ap-northeast-1.pooler.supabase.com:6543). Additionally, structuring queries in Node.js required strict adherence to positional placeholders ($1, $2), as the node-postgres driver rejects string concatenation and requires array-bound parameters.

Through this implementation, we acquired key insights into safely linking an enterprise user interface with a relational database. Most significantly, we realized why the client-side browser must never connect directly to PostgreSQL: embedding database credentials or exposing port 5432 directly to the web creates catastrophic security vulnerabilities, leaking connection credentials and permitting arbitrary SQL injection. By placing an Express REST API backend between the browser and PostgreSQL, we enforced parameterized queries that completely neutralize SQL injection, kept credentials secured within .env and excluded from GitHub via .gitignore, and implemented Role-Based Access Control (RBAC) ensuring litigants and advocates access only their assigned cases while administrative modifications remain strictly restricted to verified court registry staff.
